<script>

let boxCount = 30;

/* =========================

CREATE ROW

========================= */

function createRow(number){

const tr = document.createElement("tr");

tr.dataset.box = number;

tr.innerHTML =

"<td>" + number + "</td>" +

'<td>' +

'<input type="number" min="0" step="1" ' +

'inputmode="numeric" class="qty" ' +

'data-box="' + number + '" ' +

'oninput="calculate()">' +

'</td>' +

'<td>' +

'<input type="number" min="0" step="0.01" ' +

'inputmode="decimal" class="weight" ' +

'data-box="' + number + '" ' +

'oninput="calculate()">' +

'</td>';

return tr;

}

/* =========================

RENDER BOXES

========================= */

function renderRows(){

const left = document.getElementById("leftRows");

const right = document.getElementById("rightRows");

left.innerHTML = "";

right.innerHTML = "";

for(let i = 1; i <= boxCount; i++){

if(i <= 15){

left.appendChild(createRow(i));

}

else{

right.appendChild(createRow(i));

}

}

calculate();

}

/* =========================

CALCULATE TOTAL

========================= */

function calculate(){

let leftQ = 0;

let leftW = 0;

let rightQ = 0;

let rightW = 0;

document.querySelectorAll(".qty").forEach(q => {

const box = Number(q.dataset.box);

const weightInput =

document.querySelector(

'.weight[data-box="' + box + '"]'

);

const qv = parseFloat(q.value) || 0;

const wv = parseFloat(weightInput.value) || 0;

if(box <= 15){

leftQ += qv;

leftW += wv;

}

else{

rightQ += qv;

rightW += wv;

}

});

const totalQ = leftQ + rightQ;

const totalW = leftW + rightW;

document.getElementById("leftQty").textContent =

leftQ;

document.getElementById("leftWeight").textContent =

leftW.toFixed(2);

document.getElementById("rightQty").textContent =

rightQ;

document.getElementById("rightWeight").textContent =

rightW.toFixed(2);

document.getElementById("totalQty").textContent =

totalQ;

document.getElementById("totalWeight").textContent =

totalW.toFixed(2);

document.getElementById("totalAvg").textContent =

totalQ

? (totalW / totalQ).toFixed(2)

: "0.00";

}

/* =========================

ADD BOX

========================= */

function addBox(){

boxCount++;

const rightRows =

document.getElementById("rightRows");

rightRows.appendChild(

createRow(boxCount)

);

calculate();

}

/* =========================

REMOVE BOX

========================= */

function removeBox(){

if(boxCount <= 1){

alert(

"कम से कम 1 Box रखना जरूरी है."

);

return;

}

const row =

document.querySelector(

'tr[data-box="' + boxCount + '"]'

);

if(row){

row.remove();

}

boxCount--;

calculate();

}

function newBill(){

!confirm(

"New Bill बनाने से सभी Qty और Weight खाली हो जाएंगे. Continue?"

)

){

return;

}

boxCount = 30;

document.getElementById(

"billDate"

).value = "";

renderRows();

window.scrollTo({

top:0,

behavior:"smooth"

});

}

/* =========================

DATE

========================= */

function getDate(){

const v =

document.getElementById(

"billDate"

).value;

if(!v){

return "-";

}

const p = v.split("-");

return (

p[2] +

"-" +

p[1] +

"-" +

p[0]

);

}

/* =========================

BILL TEXT

========================= */

function billText(){

calculate();

let lines = [

"WEIGHT BILL",

"Date: " + getDate(),

"--------------------"

];

document

.querySelectorAll(

'tr[data-box]'

)

.forEach(row => {

const box =

row.dataset.box;

const q =

row.querySelector(

".qty"

).value || "0";

const w =

row.querySelector(

".weight"

).value || "0";

if(q !== "0" || w !== "0"){

lines.push(

"Box " +

box +

": Qty " +

q +

" | Weight " +

w

);

}

});

lines.push(

"--------------------"

);

lines.push(

"Total Qty: " +

document.getElementById(

"totalQty"

).textContent

);

lines.push(

"Total Weight: " +

document.getElementById(

"totalWeight"

).textContent

);

lines.push(

"Total Avg: " +

document.getElementById(

"totalAvg"

).textContent

);

return lines.join("\n");

}

/* =========================

LOAD HTML2CANVAS

========================= */

function loadHtml2Canvas(){

return new Promise(

(resolve,reject) => {

if(window.html2canvas){

resolve();

return;

}

const s =

document.createElement(

"script"

);

s.src =

"https://cdnjs.cloudflare.com/ajax/libs/html2canvas/1.4.1/html2canvas.min.js";

s.onload = resolve;

s.onerror = reject;

document.head.appendChild(s);

}

);

}

/* =========================

MAKE JPG CANVAS

========================= */

async function makeJPG(){

prepareExport();

await loadHtml2Canvas();

return await html2canvas(

document.getElementById(

"billExport"

),

{

scale:3,

useCORS:true,

backgroundColor:"#ffffff",

logging:false

}

);

}

/* =========================

SAVE JPG

========================= */

async function saveJPG(){

try{

const canvas =

await makeJPG();

const link =

document.createElement(

"a"

);

link.download =

"Weight_Bill_" +

new Date()

.toISOString()

.slice(0,10) +

".jpg";

link.href =

canvas.toDataURL(

"image/jpeg",

0.95

);

link.click();

}

catch(e){

alert(

"JPG banane me problem aayi. Internet ON karke dobara try kare."

);

}

}

/* =========================

SHARE

========================= */

async function shareBill(){

try{

const canvas =

await makeJPG();

const blob =

await new Promise(

resolve =>

canvas.toBlob(

resolve,

"image/jpeg",

0.95

)

);

const file =

new File(

[blob],

"Weight_Bill.jpg",

{

type:"image/jpeg"

}

);

if(

navigator.share &&

navigator.canShare &&

navigator.canShare({

files:[file]

})

){

await navigator.share({

title:"Weight Bill",

files:[file]

});

}

else if(navigator.share){

await navigator.share({

title:"Weight Bill",

text:billText()

});

}

else{

window.open(

"https://wa.me/?text=" +

encodeURIComponent(

billText()

),

"_blank"

);

}

}

catch(e){

if(

e.name !== "AbortError"

){

window.open(

"https://wa.me/?text=" +

encodeURIComponent(

billText()

),

"_blank"

);

}

}

}
